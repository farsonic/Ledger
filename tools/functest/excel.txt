# Where Ledger follows Excel and LibreOffice (the reference the suite checks against) doesn't.
# One line each: the formula as the test writes it (literal form or cells form), then why.
=_xlfn.CONCAT("a",1,TRUE)	Excel joins TRUE as "TRUE"; LibreOffice as 1
=_xlfn.CONCAT(E48,F48,G48)	the same, from cells
=COUNT(E56,F56,G56,H56)	Excel doesn't count text in cells, even "2"
=COUNT({1;"2";"x";TRUE;5})	Excel counts only the numbers in an array
=COUNT(Data!A95:A99)	nor text or TRUE in a range
=DOLLAR(-1234.567,1)	Excel's own example shows negatives in brackets
=DOLLAR(E80,F80)	the same, from cells
=LARGE({3;1;4;1;5;9;2;6},9)	Excel: #NUM! when k is past the end
=LARGE(Data!A193:A200,E162)	the same, from cells
=LN(0)	Excel: #NUM!
=LN(E171)	the same, from cells
=MATCH("b*",{"apple";"banana";"cherry";"apple"},0)	Excel matches wildcards in arrays too
=MROUND(10,-3)	Excel: #NUM! when the signs differ
=MROUND(E206,F206)	the same, from cells
=_xlfn.NUMBERVALUE("3.5%")	Excel reads a trailing % as a percentage
=_xlfn.NUMBERVALUE(E221)	the same, from cells
=_xlfn.NUMBERVALUE("12.5%%")	each % divides by 100 in Excel
=_xlfn.NUMBERVALUE(E373)	the same, from cells
=_xlfn.NUMBERVALUE("")	Excel: empty text is 0
=_xlfn.NUMBERVALUE(E375)	the same, from cells
=POWER(-8,1/3)	Excel: #NUM! for a fractional power of a negative
=POWER(E239,F239)	the same, from cells
=SQRT(-1)	Excel: #NUM!
=SQRT(E287)	the same, from cells
=FLOOR(2.5,-2)	Excel: #NUM! for a positive number and negative significance
=FLOOR(E368,F368)	the same, from cells
=SUM("3",TRUE)	Excel adds text numbers and logicals typed into the call
=SUM(E297,F297)	but not those in cells
=SUM({1;"3";TRUE})	nor those in an array
=SUM(Data!A459:A461)	nor in a range
=TYPE(TRUE)	Excel: 4 for a logical
=TYPE(E332)	the same, from a cell
=_xlfn.XLOOKUP(3,{1;2;3},{"a";"b";"c"})	this LibreOffice has no XLOOKUP: checked by hand
=_xlfn.XLOOKUP(E355,Data!A583:A585,Data!A587:A589)	(checked by hand)
=_xlfn.XLOOKUP(9,{1;2;3},{"a";"b";"c"},"none")	(checked by hand)
=_xlfn.XLOOKUP(E356,Data!A591:A593,Data!A595:A597,F356)	(checked by hand)
=_xlfn.XLOOKUP(2.5,{1;2;3},{"a";"b";"c"},"none",1)	(checked by hand)
=_xlfn.XLOOKUP(E357,Data!A599:A601,Data!A603:A605,F357,G357)	(checked by hand)
# literal and cells forms differ on purpose
LIT:COUNT	typed-in text numbers and logicals count; in cells they don't
LIT:COUNTA	"" typed in is a value; an empty cell isn't
LIT:ISBLANK	"" typed in isn't blank; an empty cell is
LIT:SUM	as COUNT
